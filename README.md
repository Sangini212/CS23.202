class Number {
    int a;
    int b;
    void input(int x, int y) {
        a = x;
        b = y;
    }
    void output() {
        System.out.println("First Number = " + a);
        System.out.println("Second Number = " + b);
    }
}
public class Main {
    public static void main(String[] args) {
        Number n1 = new Number();
        n1.input(10, 20);
        n1.output();

    }
}
